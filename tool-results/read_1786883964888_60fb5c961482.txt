    54→    54→// Each flow stores { action, payload }. Cleared on any navigation.
    55→    55→type UserState =
    56→    56→  | { action: "edit_price"; slug: string }

    64→    64→// ── Helper: extract callback data from ctx ───────────────────────────
    65→    65→// grammy's ctx.match is a RegExpMatchArray when using regex patterns.
    66→    66→// We need the raw callback_data string instead, which is in ctx.callbackQuery.data.
    67→    67→const cbData = (ctx: Context): string => ctx.callbackQuery?.data || "";
    68→    68→
    69→    69→// Extract the "payload" portion of a callback data string (everything after the first colon).
    70→    70→// Example: "o:HN-12345" → "HN-12345"; "oss:HN-12345:confirmed" → "HN-12345:confirmed"
    71→    71→const cbPayload = (ctx: Context): string => {
    72→    72→  const data = cbData(ctx);
    73→    73→  const idx = data.indexOf(":");
    74→    74→  return idx >= 0 ? data.slice(idx + 1) : "";
    75→    75→};
    76→    76→
    77→    77→// Extract page number from a paginated callback like "today:0", "all:2", "cust:1"
    78→    78→const cbPage = (ctx: Context): number => {
    79→    79→  const data = cbData(ctx);
    80→    80→  const parts = data.split(":");
    81→    81→  return Number(parts[1] || 0);
    82→    82→};

    85→    85→async function show(ctx: Context, text: string, keyboard?: any) {
    86→    86→  const opts: any = {
    87→    87→    parse_mode: "HTML",
    88→    88→    disable_web_page_preview: true,
    89→    89→  };
    90→    90→  if (keyboard) opts.reply_markup = keyboard;
    91→    91→  try {
    92→    92→    await ctx.editMessageText(text, opts);
    93→    93→  } catch {
    94→    94→    try {
    95→    95→      await ctx.reply(text, opts);
    96→    96→    } catch (e) {
    97→    97→      console.error("show() reply failed:", e);
    98→    98→    }
    99→    99→  }
   100→   100→}

   104→   104→  if (!ctx.from) return;

   115→   115→}

   117→   117→// Auto-answer callback queries to remove loading spinner.
   118→   118→// ALSO: clear any pending multi-step flow state (edit_price / edit_desc) on
   119→   119→// every button click. This prevents a stale state from causing the admin's next
   120→   120→// text message to be misinterpreted as a price/desc edit for the OLD product.
   121→   121→// (Handlers that START a flow — handleProductEditPrice, handleProductEditDesc —
   122→   122→// re-set the state AFTER this middleware clears it, so the flow still works.)
   123→   123→export async function answerCallbacks(ctx: Context, next: () => Promise<void>) {
   124→   124→  if (ctx.callbackQuery) {
   125→   125→    try {
   126→   126→      await ctx.answerCallbackQuery();
   127→   127→    } catch {}
   128→   128→    // Clear any pending edit flow — the admin clicked a button, so they're
   129→   129→    // navigating away from the text-input flow. If they clicked "edit price"
   130→   130→    // or "edit desc", the handler will re-set the state after this clears it.
   131→   131→    if (ctx.from) clearState(ctx.from.id);
   132→   132→  }
   133→   133→  return next();
   134→   134→}

   322→   322→// ── Set order status ─────────────────────────────────────────────────
   323→   323→// This is the admin's primary manual action — changing an order's status.
   324→   324→// It handles the full status workflow: awaiting_payment → paid → confirmed
   325→   325→// → preparing → shipped → delivered (plus cancelled as a side state).
   326→   326→// When the admin advances to "confirmed" or later, payment is also marked
   327→   327→// as confirmed. When moved back to "awaiting_payment", payment resets to pending.
   328→   328→//
   329→   329→// SPECIAL CASE: When the admin advances to "shipped" (تحویل به پست), the
   330→   330→// bot asks for the post tracking code BEFORE updating the status. The admin
   331→   331→// either sends the code as a text message (which is then saved along with
   332→   332→// the status) OR taps "بدون کد رهگیری" to skip and just set the status.
   333→   333→export async function handleSetOrderStatus(ctx: Context) {
   334→   334→  // pattern: oss:<orderNumber>:<status>  (orderNumber has no colon, so parts[1]=orderNumber, parts[2]=status)
   335→   335→  const parts = cbData(ctx).split(":");
   336→   336→  const orderNumber = parts[1];
   337→   337→  const status = parts[2];
   338→   338→
   339→   339→  // Validate the status is a known value
   340→   340→  const validStatuses = [
   341→   341→    "awaiting_payment", "paid", "confirmed", "preparing", "shipped", "delivered", "cancelled",
   342→   342→  ];
   343→   343→  if (!validStatuses.includes(status)) {
   344→   344→    await show(ctx, "❌ وضعیت نامعتبر است.", backKb());
   345→   345→    return;
   346→   346→  }
   347→   347→
   348→   348→  try {
   349→   349→    // Show a "processing" indicator so the admin knows the click was received
   350→   350→    try {
   351→   351→      await ctx.editMessageText("⏳ در حال به‌روزرسانی وضعیت...", { parse_mode: "HTML" });
   352→   352→    } catch {}
   353→   353→
   354→   354→    const order = await withRetry(
   355→   355→      () => db.order.findUnique({
   356→   356→        where: { orderNumber },
   357→   357→        select: {
   358→   358→          id: true,
   359→   359→          customerName: true,
   360→   360→          orderStatus: true,
   361→   361→          deliveryType: true,
   362→   362→          trackingCode: true,
   363→   363→        },
   364→   364→      }),
   365→   365→      { label: `findOrder(${orderNumber})` }
   366→   366→    );
   367→   367→    if (!order) {
   368→   368→      await show(ctx, "❌ سفارش یافت نشد.", backKb());
   369→   369→      return;
   370→   370→    }
   371→   371→
   372→   372→    // If status hasn't changed, no-op (avoid unnecessary writes)
   373→   373→    if (order.orderStatus === status) {
   374→   374→      const text = await orderDetailsMessage(orderNumber);
   375→   375→      await show(ctx, `ℹ️ وضعیت سفارش از قبل «${statusLabel(status)}» بود.\n\n${text || ""}`, orderActionsKb(orderNumber, status));
   376→   376→      return;
   377→   377→    }
   378→   378→
   379→   379→    // SPECIAL CASE: status === "shipped" → prompt for the post tracking code
   380→   380→    // before updating. The admin enters the code (or skips) and the order is
   381→   381→    // updated in the tracking-code handler.
   382→   382→    if (status === "shipped") {

   442→   442→    const text = await orderDetailsMessage(orderNumber);

   444→   444→      `✅ <b>وضعیت سفارش به‌روزرسانی شد</b>\n\n` +

   716→   716→// ── Text message handler (search + price/desc edit) ──────────────────
   717→   717→export async function handleTextMessage(ctx: Context) {
   718→   718→  const text = ctx.message?.text || "";
   719→   719→  const userId = ctx.from!.id;
   720→   720→
   721→   721→  // Check if in a multi-step flow
   722→   722→  const state = userState.get(userId);
   723→   723→  if (state && state.action === "edit_price") {
   724→   724→    clearState(userId);
   725→   725→    // Convert Persian/Arabic digits to ASCII, then validate.
   726→   726→    // IMPORTANT: We must reject negative numbers, decimals, and any non-digit
   727→   727→    // characters. The old code used .replace(/[^\d]/g, "") which STRIPPED
   728→   728→    // the minus sign, turning "-100" into "100" (a valid positive price!).
   729→   729→    // Now we require the input to be purely digits (after whitespace removal).
   730→   730→    const asciiText = toAsciiDigits(text).replace(/\s/g, "");
   731→   731→    if (!/^\d+$/.test(asciiText)) {
   732→   732→      await ctx.reply(
   733→   733→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید (فقط ارقام، بدون علامت یا کاما).",
   734→   734→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   735→   735→      );
   736→   736→      return;
   737→   737→    }
   738→   738→    const price = parseInt(asciiText, 10);
   739→   739→    if (isNaN(price) || price <= 0) {
   740→   740→      await ctx.reply(
   741→   741→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید.",
   742→   742→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   743→   743→      );
   744→   744→      return;
   745→   745→    }
   746→   746→    if (price > 1_000_000_000) {
   747→   747→      await ctx.reply(
   748→   748→        "❌ قیمت بیش از حد بزرگ است. لطفاً مقدار را بررسی کنید.",
   749→   749→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   750→   750→      );
   751→   751→      return;
   752→   752→    }
   753→   753→    // Fetch old price for the diff message
   754→   754→    const before = await db.product.findUnique({
   755→   755→      where: { slug: state.slug },
   756→   756→      select: { name: true, pricePerKg: true, featured: true },
   757→   757→    });
   758→   758→    if (!before) {
   759→   759→      await ctx.reply("❌ محصول یافت نشد.", backKb());
   760→   760→      return;
   761→   761→    }
   762→   762→    await db.product.update({
   763→   763→      where: { slug: state.slug },
   764→   764→      data: { pricePerKg: price },
   765→   765→    });
   766→   766→    const detailText = await productDetailsMessage(state.slug);
   767→   767→    const diff = price - (before.pricePerKg || 0);
   768→   768→    const diffStr =
   769→   769→      diff === 0
   770→   770→        ? "بدون تغییر"
   771→   771→        : diff > 0
   772→   772→        ? `🔺 افزایش: +${formatToman(diff)}`
   773→   773→        : `🔻 کاهش: ${formatToman(Math.abs(diff))}`;
   774→   774→    await ctx.reply(
   775→   775→      `✅ <b>قیمت محصول به‌روزرسانی شد</b>\n\n` +
   776→   776→        `🍯 ${escapeHtml(before.name)}\n` +
   777→   777→        `💰 قیمت قبلی: ${formatToman(before.pricePerKg)}\n` +
   778→   778→        `💰 قیمت جدید: <b>${formatToman(price)}</b>\n` +
   779→   779→        `📊 ${diffStr}\n\n` +
   780→   780→        `🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   781→   781→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, before.featured) }
   782→   782→    );
   783→   783→    console.log(
   784→   784→      `💰 Product price: ${state.slug} ${before.pricePerKg} → ${price}`
   785→   785→    );
   786→   786→    return;
   787→   787→  }
   788→   788→
   789→   789→  if (state && state.action === "edit_desc") {
   790→   790→    clearState(userId);
   791→   791→    if (!text.trim()) {
   792→   792→      await ctx.reply(
   793→   793→        "❌ توضیحات نمی‌تواند خالی باشد.",
   794→   794→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   795→   795→      );
   796→   796→      return;
   797→   797→    }
   798→   798→    await db.product.update({
   799→   799→      where: { slug: state.slug },
   800→   800→      data: { description: text.trim() },
   801→   801→    });
   802→   802→    const product = await db.product.findUnique({
   803→   803→      where: { slug: state.slug },
   804→   804→      select: { name: true, featured: true },
   805→   805→    });
   806→   806→    const detailText = await productDetailsMessage(state.slug);
   807→   807→    await ctx.reply(
   808→   808→      `✅ توضیحات محصول «${escapeHtml(product?.name || "")}» به‌روزرسانی شد.\n\n🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   809→   809→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, product?.featured || false) }
   810→   810→    );
   811→   811→    console.log(`📝 Product desc updated: ${state.slug}`);
   812→   812→    return;
   813→   813→  }
   814→   814→
   815→   815→  // Tracking code entry flow — when admin is setting an order to "shipped"
   816→   816→  // (تحویل به پست), they must enter the post tracking code.
   817→   817→  if (state && state.action === "enter_tracking") {
   818→   818→    const orderNumber = state.orderNumber;
   819→   819→    clearState(userId);
   820→   820→    // Normalize: convert Persian/Arabic digits to ASCII, strip whitespace.
   821→   821→    // Tracking codes are typically all digits (13-20 digits for Iran Post),
   822→   822→    // but we also allow letters (some registered post uses letters).
   823→   823→    const normalized = toAsciiDigits(text).replace(/\s+/g, "");
   824→   824→    // Validation: length 8-30, only alphanumeric
   825→   825→    if (!/^[A-Za-z0-9]{8,30}$/.test(normalized)) {
   826→   826→      await ctx.reply(
   827→   827→        "❌ کد رهگیری نامعتبر است.\n\n" +
   828→   828→          "کد رهگیری باید:\n" +
   829→   829→          "• بین ۸ تا ۳۰ نویسه باشد\n" +
   830→   830→          "• فقط شامل ارقام و حروف انگلیسی باشد\n" +
   831→   831→          "• بدون فاصله یا کاراکتر خاص باشد\n\n" +
   832→   832→          "لطفاً کد رهگیری صحیح را ارسال کنید یا روی «بدون کد رهگیری» بزنید.",
   833→   833→        { parse_mode: "HTML", reply_markup: trackingEntryKb(orderNumber) }
   834→   834→      );
   835→   835→      // Re-set state so the next message is treated as another attempt
   836→   836→      userState.set(userId, { action: "enter_tracking", orderNumber });
   837→   837→      return;
   838→   838→    }
   839→   839→    try {
   840→   840→      const order = await withRetry(
   841→   841→        () => db.order.findUnique({
   842→   842→          where: { orderNumber },
   843→   843→          select: { id: true, customerName: true, orderStatus: true },
   844→   844→        }),
   845→   845→        { label: `findOrderTracking(${orderNumber})` }
   846→   846→      );
   847→   847→      if (!order) {
   848→   848→        await ctx.reply("❌ سفارش یافت نشد.", { reply_markup: backKb(), parse_mode: "HTML" });
   849→   849→        return;
   850→   850→      }
   851→   851→      await withRetry(
   852→   852→        () => db.order.update({
   853→   853→          where: { id: order.id },
   854→   854→          data: {
   855→   855→            orderStatus: "shipped",
   856→   856→            paymentStatus: "confirmed",
   857→   857→            trackingCode: normalized,
   858→   858→          },
   859→   859→        }),
   860→   860→        { label: `updateOrderTracking(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
   861→   861→      );
   862→   862→      console.log(
   863→   863→        `📊 Status change: ${orderNumber} ${order.orderStatus} → shipped (tracking: ${normalized})`
   864→   864→      );
   865→   865→      const detailText = await orderDetailsMessage(orderNumber);
   866→   866→      await ctx.reply(
   867→   867→        `✅ <b>سفارش به پست تحویل داده شد</b>\n\n` +
   868→   868→          `📋 <code>${orderNumber}</code>\n` +
   869→   869→          `👤 ${escapeHtml(order.customerName)}\n` +
   870→   870→          `📊 وضعیت قبلی: ${statusLabel(order.orderStatus)}\n` +
   871→   871→          `📊 وضعیت جدید: ${statusLabel("shipped")}\n` +
   872→   872→          `📮 کد رهگیری: <code>${escapeHtml(normalized)}</code>\n\n` +
   873→   873→          `🌐 مشتری می‌تواند با این کد رهگیری، وضعیت لحظه‌ای بسته را در سامانه پست پیگیری کند.\n\n` +
   874→   874→          (detailText || ""),
   875→   875→        { parse_mode: "HTML", reply_markup: orderActionsKb(orderNumber, "shipped") }
   876→   876→      );
   877→   877→    } catch (e: any) {
   878→   878→      const errCode = e?.code || e?.errno || "N/A";
   879→   879→      const errMsg = String(e?.message || e).slice(0, 300);
   880→   880→      console.error(
   881→   881→        `❌ enter_tracking FAILED for ${orderNumber} | code=${errCode} | msg=${errMsg}`
   882→   882→      );
   883→   883→      await ctx.reply(
   884→   884→        `❌ خطا در ثبت کد رهگیری.\n\n⚠️ خطا: <code>${errCode}</code>\n\nلطفاً دوباره تلاش کنید.`,
   885→   885→        { parse_mode: "HTML", reply_markup: backKb() }
   886→   886→      );
   887→   887→    }
   888→   888→    return;
   889→   889→  }
   890→   890→
   891→   891→  // Default: treat as search
   892→   892→  const query = text.trim();
   893→   893→  if (!query) return;
   894→   894→  try {
   895→   895→    await ctx.reply("⏳ در حال جستجو...", { parse_mode: "HTML" });
   896→   896→  } catch {}
   897→   897→  const msg = await searchMessage(query);
   898→   898→  const orders = await searchOrders(query);
   899→   899→  let kb = backKb();
   900→   900→  if (orders.length > 0) {
   901→   901→    kb = new InlineKeyboard();
   902→   902→    for (const o of orders) {
   903→   903→      kb.text(`${o.orderNumber} | ${o.customerName}`, `o:${o.orderNumber}`).row();
   904→   904→    }
   905→   905→    kb.text("🔙 منوی اصلی", "back");
   906→   906→  }
   907→   907→  await ctx.reply(msg, {
   908→   908→    parse_mode: "HTML",
   909→   909→    reply_markup: kb,
   910→   910→    disable_web_page_preview: true,
   911→   911→  });
   912→   912→}

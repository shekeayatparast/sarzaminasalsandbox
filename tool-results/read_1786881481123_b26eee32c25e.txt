    62→    62→// ── Helper: extract callback data from ctx ───────────────────────────
    63→    63→// grammy's ctx.match is a RegExpMatchArray when using regex patterns.
    64→    64→// We need the raw callback_data string instead, which is in ctx.callbackQuery.data.
    65→    65→const cbData = (ctx: Context): string => ctx.callbackQuery?.data || "";
    66→    66→
    67→    67→// Extract the "payload" portion of a callback data string (everything after the first colon).
    68→    68→// Example: "o:HN-12345" → "HN-12345"; "oss:HN-12345:confirmed" → "HN-12345:confirmed"
    69→    69→const cbPayload = (ctx: Context): string => {
    70→    70→  const data = cbData(ctx);
    71→    71→  const idx = data.indexOf(":");
    72→    72→  return idx >= 0 ? data.slice(idx + 1) : "";
    73→    73→};
    74→    74→
    75→    75→// Extract page number from a paginated callback like "today:0", "all:2", "cust:1"
    76→    76→const cbPage = (ctx: Context): number => {
    77→    77→  const data = cbData(ctx);
    78→    78→  const parts = data.split(":");
    79→    79→  return Number(parts[1] || 0);
    80→    80→};

    82→    82→// ── Helper: show text (edit message if possible, else reply) ─────────
    83→    83→async function show(ctx: Context, text: string, keyboard?: any) {
    84→    84→  const opts: any = {
    85→    85→    parse_mode: "HTML",
    86→    86→    disable_web_page_preview: true,
    87→    87→  };
    88→    88→  if (keyboard) opts.reply_markup = keyboard;
    89→    89→  try {
    90→    90→    await ctx.editMessageText(text, opts);
    91→    91→  } catch {
    92→    92→    try {
    93→    93→      await ctx.reply(text, opts);
    94→    94→    } catch (e) {
    95→    95→      console.error("show() reply failed:", e);
    96→    96→    }
    97→    97→  }
    98→    98→}

   309→   309→// ── Set order status ─────────────────────────────────────────────────
   310→   310→// This is the admin's primary manual action — changing an order's status.
   311→   311→// It handles the full status workflow: awaiting_payment → paid → confirmed
   312→   312→// → preparing → shipped → delivered (plus cancelled as a side state).
   313→   313→// When the admin advances to "confirmed" or later, payment is also marked
   314→   314→// as confirmed. When moved back to "awaiting_payment", payment resets to pending.
   315→   315→export async function handleSetOrderStatus(ctx: Context) {
   316→   316→  // pattern: oss:<orderNumber>:<status>  (orderNumber has no colon, so parts[1]=orderNumber, parts[2]=status)
   317→   317→  const parts = cbData(ctx).split(":");
   318→   318→  const orderNumber = parts[1];
   319→   319→  const status = parts[2];
   320→   320→
   321→   321→  // Validate the status is a known value
   322→   322→  const validStatuses = [
   323→   323→    "awaiting_payment", "paid", "confirmed", "preparing", "shipped", "delivered", "cancelled",
   324→   324→  ];
   325→   325→  if (!validStatuses.includes(status)) {
   326→   326→    await show(ctx, "❌ وضعیت نامعتبر است.", backKb());
   327→   327→    return;
   328→   328→  }
   329→   329→
   330→   330→  try {
   331→   331→    // Show a "processing" indicator so the admin knows the click was received
   332→   332→    try {
   333→   333→      await ctx.editMessageText("⏳ در حال به‌روزرسانی وضعیت...", { parse_mode: "HTML" });
   334→   334→    } catch {}
   335→   335→
   336→   336→    const order = await withRetry(
   337→   337→      () => db.order.findUnique({
   338→   338→        where: { orderNumber },
   339→   339→        select: { id: true, customerName: true, orderStatus: true },
   340→   340→      }),
   341→   341→      { label: `findOrder(${orderNumber})` }
   342→   342→    );
   343→   343→    if (!order) {
   344→   344→      await show(ctx, "❌ سفارش یافت نشد.", backKb());
   345→   345→      return;
   346→   346→    }
   347→   347→
   348→   348→    // If status hasn't changed, no-op (avoid unnecessary writes)
   349→   349→    if (order.orderStatus === status) {
   350→   350→      const text = await orderDetailsMessage(orderNumber);
   351→   351→      await show(ctx, `ℹ️ وضعیت سفارش از قبل «${statusLabel(status)}» بود.\n\n${text || ""}`, orderActionsKb(orderNumber, status));
   352→   352→      return;
   353→   353→    }
   354→   354→
   355→   355→    // Determine the correct paymentStatus for the new order status.
   356→   356→    // Rule:
   357→   357→    //   - awaiting_payment  → payment must be "pending"  (customer hasn't paid)
   358→   358→    //   - paid              → payment must be "confirmed" (customer clicked "I paid")
   359→   359→    //   - confirmed/preparing/shipped/delivered → payment must be "confirmed"
   360→   360→    //   - cancelled         → leave unchanged (preserve for accounting/audit)
   361→   361→    //
   362→   362→    // This guarantees the site's tracking page never shows an impossible
   363→   363→    // combination like "shipped + pending payment" or "awaiting + confirmed".
   364→   364→    let newPaymentStatus: string | undefined;
   365→   365→    if (status === "awaiting_payment") {
   366→   366→      newPaymentStatus = "pending";
   367→   367→    } else if (status === "cancelled") {
   368→   368→      newPaymentStatus = undefined; // preserve existing
   369→   369→    } else {
   370→   370→      // paid, confirmed, preparing, shipped, delivered
   371→   371→      newPaymentStatus = "confirmed";
   372→   372→    }
   373→   373→
   374→   374→    await withRetry(
   375→   375→      () => db.order.update({
   376→   376→        where: { id: order.id },
   377→   377→        data: {
   378→   378→          orderStatus: status,
   379→   379→          ...(newPaymentStatus !== undefined
   380→   380→            ? { paymentStatus: newPaymentStatus }
   381→   381→            : {}),
   382→   382→        },
   383→   383→      }),
   384→   384→      { label: `updateOrderStatus(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
   385→   385→    );
   386→   386→
   387→   387→    console.log(
   388→   388→      `📊 Status change: ${orderNumber} ${order.orderStatus} → ${status}` +
   389→   389→        (newPaymentStatus !== undefined
   390→   390→          ? ` (payment: ${newPaymentStatus})`
   391→   391→          : "")
   392→   392→    );
   393→   393→
   394→   394→    const text = await orderDetailsMessage(orderNumber);
   395→   395→    const msg =

   603→   603→// ── Text message handler (search + price/desc edit) ──────────────────
   604→   604→export async function handleTextMessage(ctx: Context) {
   605→   605→  const text = ctx.message?.text || "";
   606→   606→  const userId = ctx.from!.id;
   607→   607→
   608→   608→  // Check if in a multi-step flow
   609→   609→  const state = userState.get(userId);
   610→   610→  if (state && state.action === "edit_price") {
   611→   611→    clearState(userId);
   612→   612→    const price = parseInt(toAsciiDigits(text).replace(/[^\d]/g, ""), 10);
   613→   613→    if (isNaN(price) || price <= 0) {
   614→   614→      await ctx.reply(
   615→   615→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید.",
   616→   616→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   617→   617→      );
   618→   618→      return;
   619→   619→    }
   620→   620→    if (price > 1_000_000_000) {
   621→   621→      await ctx.reply(
   622→   622→        "❌ قیمت بیش از حد بزرگ است. لطفاً مقدار را بررسی کنید.",
   623→   623→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   624→   624→      );
   625→   625→      return;
   626→   626→    }
   627→   627→    // Fetch old price for the diff message
   628→   628→    const before = await db.product.findUnique({
   629→   629→      where: { slug: state.slug },
   630→   630→      select: { name: true, pricePerKg: true, featured: true },
   631→   631→    });
   632→   632→    if (!before) {
   633→   633→      await ctx.reply("❌ محصول یافت نشد.", backKb());
   634→   634→      return;
   635→   635→    }
   636→   636→    await db.product.update({
   637→   637→      where: { slug: state.slug },
   638→   638→      data: { pricePerKg: price },
   639→   639→    });
   640→   640→    const detailText = await productDetailsMessage(state.slug);
   641→   641→    const diff = price - (before.pricePerKg || 0);
   642→   642→    const diffStr =
   643→   643→      diff === 0
   644→   644→        ? "بدون تغییر"
   645→   645→        : diff > 0
   646→   646→        ? `🔺 افزایش: +${formatToman(diff)}`
   647→   647→        : `🔻 کاهش: ${formatToman(Math.abs(diff))}`;
   648→   648→    await ctx.reply(
   649→   649→      `✅ <b>قیمت محصول به‌روزرسانی شد</b>\n\n` +
   650→   650→        `🍯 ${escapeHtml(before.name)}\n` +
   651→   651→        `💰 قیمت قبلی: ${formatToman(before.pricePerKg)}\n` +
   652→   652→        `💰 قیمت جدید: <b>${formatToman(price)}</b>\n` +
   653→   653→        `📊 ${diffStr}\n\n` +
   654→   654→        `🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   655→   655→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, before.featured) }
   656→   656→    );
   657→   657→    console.log(
   658→   658→      `💰 Product price: ${state.slug} ${before.pricePerKg} → ${price}`
   659→   659→    );
   660→   660→    return;
   661→   661→  }
   662→   662→
   663→   663→  if (state && state.action === "edit_desc") {
   664→   664→    clearState(userId);
   665→   665→    if (!text.trim()) {
   666→   666→      await ctx.reply(
   667→   667→        "❌ توضیحات نمی‌تواند خالی باشد.",
   668→   668→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   669→   669→      );
   670→   670→      return;
   671→   671→    }
   672→   672→    await db.product.update({
   673→   673→      where: { slug: state.slug },
   674→   674→      data: { description: text.trim() },
   675→   675→    });
   676→   676→    const product = await db.product.findUnique({
   677→   677→      where: { slug: state.slug },
   678→   678→      select: { name: true, featured: true },
   679→   679→    });
   680→   680→    const detailText = await productDetailsMessage(state.slug);
   681→   681→    await ctx.reply(
   682→   682→      `✅ توضیحات محصول «${escapeHtml(product?.name || "")}» به‌روزرسانی شد.\n\n🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   683→   683→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, product?.featured || false) }
   684→   684→    );
   685→   685→    console.log(`📝 Product desc updated: ${state.slug}`);
   686→   686→    return;
   687→   687→  }
   688→   688→
   689→   689→  // Default: treat as search
   690→   690→  const query = text.trim();
   691→   691→  if (!query) return;
   692→   692→  try {
   693→   693→    await ctx.reply("⏳ در حال جستجو...", { parse_mode: "HTML" });
   694→   694→  } catch {}
   695→   695→  const msg = await searchMessage(query);
   696→   696→  const orders = await searchOrders(query);
   697→   697→  let kb = backKb();
   698→   698→  if (orders.length > 0) {
   699→   699→    kb = new InlineKeyboard();
   700→   700→    for (const o of orders) {
   701→   701→      kb.text(`${o.orderNumber} | ${o.customerName}`, `o:${o.orderNumber}`).row();
   702→   702→    }
   703→   703→    kb.text("🔙 منوی اصلی", "back");
   704→   704→  }
   705→   705→  await ctx.reply(msg, {
   706→   706→    parse_mode: "HTML",
   707→   707→    reply_markup: kb,
   708→   708→    disable_web_page_preview: true,
   709→   709→  });
   710→   710→}

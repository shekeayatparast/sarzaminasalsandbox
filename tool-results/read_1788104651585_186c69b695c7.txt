   102→// ── Helpers ─────────────────────────────────────────────────────────────
   103→// Persian (۰-۹) and Arabic-Indic (٠-٩) digit tables
   104→const FA_DIGITS_ARR = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];
   105→const AR_DIGITS_ARR = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"];
   106→
   107→/** Convert Persian/Arabic digits in a string to ASCII digits. */
   108→function toEnglishDigits(input: string): string {
   109→  let out = input;
   110→  for (let i = 0; i < 10; i++) {
   111→    out = out.replace(new RegExp(FA_DIGITS_ARR[i], "g"), String(i));
   112→    out = out.replace(new RegExp(AR_DIGITS_ARR[i], "g"), String(i));
   113→  }
   114→  return out;
   115→}
   116→
   117→/** Convert a Persian-style number input to a JS number. Returns NaN if empty. */
   118→function parseNumberInput(s: string): number {
   119→  const cleaned = toEnglishDigits(s).replace(/[,٬\s]/g, "");
   120→  if (cleaned === "") return NaN;
   121→  return Number(cleaned);
   122→}
   123→
   124→/** Auto-suggest a slug from a Persian name — fallback to "product-<n>". */
   125→function slugify(name: string): string {
   126→  // Persian → ASCII-ish slug. We just lowercase, replace spaces/duplicates
   127→  // with dashes, strip non a-z0-9-. The user can edit it manually anyway.
   128→  const base = name
   129→    .trim()
   130→    .toLowerCase()
   131→    .replace(/[\s_]+/g, "-")
   132→    .replace(/[^a-z0-9-]/g, "")
   133→    .replace(/-+/g, "-")
   134→    .replace(/^-|-$/g, "");
   135→  return base || "product";
   136→}

   303→    }
   304→    const stockKg = parseNumberInput(form.stockKg || "0") || 0;
   305→    if (isNaN(stockKg) || stockKg < 0) {

   307→      return;
   308→    }

   310→    const body = {
   311→      name,
   312→      slug,
   313→      description,

   315→      pricePerKg,
   316→      agentPricePerKg,

   338→        return;
   339→      }
   340→      toast.success(data.message || "محصول ذخیره شد");
   341→      onOpenChange(false);
   342→      // Give the dialog close animation a tick, then refresh server data
   343→      setTimeout(() => router.refresh(), 250);
   344→    } catch (err) {
   345→      console.error("[ProductForm submit] error:", err);
   346→      toast.error("خطای شبکه. لطفاً دوباره تلاش کنید.");

   656→// ── Sub-component: StockEditor (inline, B10) ────────────────────────────
   657→function StockEditor({
   658→  product,
   659→  onSaved,
   660→}: {
   661→  product: AdminProduct;
   662→  onSaved?: () => void;
   663→}) {
   664→  const [value, setValue] = useState<string>(String(product.stockKg));
   665→  const [saving, setSaving] = useState(false);
   666→
   667→  // Keep local value in sync if the product row updates from server refresh
   668→  useEffect(() => {
   669→    setValue(String(product.stockKg));
   670→  }, [product.stockKg]);
   671→
   672→  const stock = parseNumberInput(value);
   673→  const dirty = !isNaN(stock) && stock !== product.stockKg;
   674→  const low = !isNaN(stock) && stock < LOW_STOCK_THRESHOLD;
   675→
   676→  const save = async () => {
   677→    if (saving || !dirty) return;
   678→    if (isNaN(stock) || stock < 0) {
   679→      toast.error("موجودی باید عدد مثبت باشد");
   680→      return;
   681→    }
   682→    setSaving(true);
   683→    try {
   684→      const res = await fetch(`/api/admin/products/${product.id}`, {
   685→        method: "PATCH",
   686→        headers: { "Content-Type": "application/json" },
   687→        body: JSON.stringify({ stockKg: stock }),
   688→      });
   689→      const data = await res.json();
   690→      if (!res.ok) {
   691→        toast.error(data.error || "ذخیره ناموفق بود");
   692→        return;
   693→      }
   694→      toast.success(`موجودی «${product.name}» به‌روزرسانی شد`);
   695→      onSaved?.();
   696→    } catch (err) {
   697→      console.error("[StockEditor save] error:", err);
   698→      toast.error("خطای شبکه. لطفاً دوباره تلاش کنید.");
   699→    } finally {
   700→      setSaving(false);
   701→    }
   702→  };
   703→
   704→  return (
   705→    <div className="flex items-center gap-1.5" dir="ltr">
   706→      <Input
   707→        type="text"
   708→        inputMode="decimal"
   709→        value={value}
   710→        onChange={(e) => setValue(e.target.value)}
   711→        disabled={saving}
   712→        className={cn(
   713→          "h-8 w-20 text-center text-sm",
   714→          low ? "text-red-600 border-red-300" : ""
   715→        )}
   716→        title="موجودی به کیلوگرم"
   717→      />
   718→      <span className="text-[10px] text-muted-foreground">کیلو</span>
   719→      <Button
   720→        type="button"
   721→        size="icon"
   722→        variant="ghost"
   723→        className={cn(
   724→          "h-7 w-7",
   725→          dirty
   726→            ? "bg-honey-light/40 text-honey-dark hover:bg-honey-light/60"
   727→            : "opacity-40"
   728→        )}
   729→        onClick={save}
   730→        disabled={saving || !dirty}
   731→        title="ذخیره موجودی"
   732→      >
   733→        {saving ? (
   734→          <Loader2 className="w-3.5 h-3.5 animate-spin" />
   735→        ) : (
   736→          <Save className="w-3.5 h-3.5" />
   737→        )}
   738→      </Button>
   739→    </div>
   740→  );
   741→}

   744→function DeleteDialog({
   745→  product,
   746→  open,
   747→  onOpenChange,
   748→  onDone,
   749→}: {
   750→  product: AdminProduct | null;
   751→  open: boolean;
   752→  onOpenChange: (v: boolean) => void;

   762→        method: "DELETE",

   782→      <AlertDialogContent dir="rtl">

   806→              deleting && "opacity-70 cursor-wait"

   810→              <Loader2 className="w-4 h-4 animate-spin" />
   811→            ) : (
   812→              <Trash2 className="w-4 h-4" />

   814→            حذف محصول
   815→          </AlertDialogAction>
   816→        </AlertDialogFooter>
   817→      </AlertDialogContent>

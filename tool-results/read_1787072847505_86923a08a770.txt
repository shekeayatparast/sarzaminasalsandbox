    55→// ── In-memory state for multi-step flows ─────────────────────────────
    56→// Each flow stores { action, payload }. Cleared on any navigation.
    57→type UserState =
    58→  | { action: "edit_price"; slug: string }
    59→  | { action: "edit_desc"; slug: string }
    60→  | { action: "enter_tracking"; orderNumber: string; isEdit?: boolean };
    61→
    62→const userState = new Map<number, UserState>();
    63→
    64→const clearState = (userId: number) => userState.delete(userId);
    65→
    66→// ── Helper: extract callback data from ctx ───────────────────────────
    67→// grammy's ctx.match is a RegExpMatchArray when using regex patterns.
    68→// We need the raw callback_data string instead, which is in ctx.callbackQuery.data.
    69→const cbData = (ctx: Context): string => ctx.callbackQuery?.data || "";
    70→
    71→// Extract the "payload" portion of a callback data string (everything after the first colon).
    72→// Example: "o:HN-12345" → "HN-12345"; "oss:HN-12345:confirmed" → "HN-12345:confirmed"
    73→const cbPayload = (ctx: Context): string => {
    74→  const data = cbData(ctx);
    75→  const idx = data.indexOf(":");
    76→  return idx >= 0 ? data.slice(idx + 1) : "";
    77→};
    78→
    79→// Extract page number from a paginated callback like "today:0", "all:2", "cust:1"
    80→const cbPage = (ctx: Context): number => {
    81→  const data = cbData(ctx);
    82→  const parts = data.split(":");
    83→  return Number(parts[1] || 0);
    84→};

    86→// ── Helper: show text (edit message if possible, else reply) ─────────
    87→async function show(ctx: Context, text: string, keyboard?: any) {
    88→  const opts: any = {
    89→    parse_mode: "HTML",
    90→    disable_web_page_preview: true,
    91→  };
    92→  if (keyboard) opts.reply_markup = keyboard;
    93→  try {
    94→    await ctx.editMessageText(text, opts);
    95→  } catch {
    96→    try {
    97→      await ctx.reply(text, opts);
    98→    } catch (e) {
    99→      console.error("show() reply failed:", e);
   100→    }
   101→  }
   102→}

   104→// ── Access control middleware ────────────────────────────────────────
   105→export async function accessControl(ctx: Context, next: () => Promise<void>) {
   106→  if (!ctx.from) return;
   107→  if (ctx.from.id !== ADMIN_ID) {
   108→    try {
   109→      await ctx.reply(
   110→        "⛔ شما دسترسی به این ربات ندارید.\n\n" +
   111→          "این ربات اختصاصاً برای مدیریت فروشگاه «سرزمین عسل» طراحی شده است."
   112→      );
   113→    } catch {}
   114→    return;
   115→  }
   116→  return next();
   117→}

   119→// Auto-answer callback queries to remove loading spinner.
   120→// ALSO: clear any pending multi-step flow state (edit_price / edit_desc) on
   121→// every button click. This prevents a stale state from causing the admin's next
   122→// text message to be misinterpreted as a price/desc edit for the OLD product.
   123→// (Handlers that START a flow — handleProductEditPrice, handleProductEditDesc —
   124→// re-set the state AFTER this middleware clears it, so the flow still works.)
   125→export async function answerCallbacks(ctx: Context, next: () => Promise<void>) {
   126→  if (ctx.callbackQuery) {
   127→    try {
   128→      await ctx.answerCallbackQuery();
   129→    } catch {}
   130→    // Clear any pending edit flow — the admin clicked a button, so they're
   131→    // navigating away from the text-input flow. If they clicked "edit price"
   132→    // or "edit desc", the handler will re-set the state after this clears it.
   133→    if (ctx.from) clearState(ctx.from.id);
   134→  }
   135→  return next();
   136→}

   141→async function fetchMainMenuStats(): Promise<{ todayCount: number; verifyCount: number }> {
   142→  // "today" is defined as midnight Iran time (Asia/Tehran), NOT server-local UTC.
   143→  const todayStart = startOfTodayIran();
   144→  const [todayCount, verifyCount] = await Promise.all([
   145→    db.order.count({ where: { createdAt: { gte: todayStart } } }),
   146→    db.order.count({ where: { orderStatus: "paid" } }),
   147→  ]);
   148→  return { todayCount, verifyCount };
   149→}

   327→// ── Set order status ─────────────────────────────────────────────────
   328→// This is the admin's primary manual action — changing an order's status.
   329→// It handles the full status workflow: awaiting_payment → paid → confirmed
   330→// → preparing → shipped → delivered (plus cancelled as a side state).
   331→// When the admin advances to "confirmed" or later, payment is also marked
   332→// as confirmed. When moved back to "awaiting_payment", payment resets to pending.
   333→//
   334→// SPECIAL CASE: When the admin advances to "shipped" (تحویل به پست), the
   335→// bot asks for the post tracking code BEFORE updating the status. The admin
   336→// either sends the code as a text message (which is then saved along with
   337→// the status) OR taps "بدون کد رهگیری" to skip and just set the status.
   338→export async function handleSetOrderStatus(ctx: Context) {
   339→  // pattern: oss:<orderNumber>:<status>  (orderNumber has no colon, so parts[1]=orderNumber, parts[2]=status)
   340→  const parts = cbData(ctx).split(":");
   341→  const orderNumber = parts[1];
   342→  const status = parts[2];
   343→
   344→  // Validate the status is a known value
   345→  const validStatuses = [
   346→    "awaiting_payment", "paid", "confirmed", "preparing", "shipped", "delivered", "cancelled",
   347→  ];
   348→  if (!validStatuses.includes(status)) {
   349→    await show(ctx, "❌ وضعیت نامعتبر است.", backKb());
   350→    return;
   351→  }
   352→
   353→  try {
   354→    // Show a "processing" indicator so the admin knows the click was received
   355→    try {
   356→      await ctx.editMessageText("⏳ در حال به‌روزرسانی وضعیت...", { parse_mode: "HTML" });
   357→    } catch {}
   358→
   359→    const order = await withRetry(
   360→      () => db.order.findUnique({
   361→        where: { orderNumber },
   362→        select: {
   363→          id: true,
   364→          customerName: true,
   365→          orderStatus: true,
   366→          deliveryType: true,
   367→          trackingCode: true,
   368→        },
   369→      }),
   370→      { label: `findOrder(${orderNumber})` }
   371→    );
   372→    if (!order) {
   373→      await show(ctx, "❌ سفارش یافت نشد.", backKb());
   374→      return;
   375→    }
   376→
   377→    // If status hasn't changed, no-op (avoid unnecessary writes)
   378→    if (order.orderStatus === status) {
   379→      const text = await orderDetailsMessage(orderNumber);
   380→      await show(
   381→        ctx,
   382→        `ℹ️ وضعیت سفارش از قبل «${statusLabel(status)}» بود.\n\n${text || ""}`,
   383→        orderActionsKb(orderNumber, status, !!order.trackingCode)
   384→      );
   385→      return;
   386→    }
   387→
   388→    // SPECIAL CASE: status === "shipped" → prompt for the post tracking code
   389→    // before updating. The admin enters the code (or skips) and the order is
   390→    // updated in the tracking-code handler.
   391→    if (status === "shipped") {
   392→      // Re-set the state AFTER the answerCallbacks middleware cleared it.
   393→      userState.set(ctx.from!.id, { action: "enter_tracking", orderNumber });
   394→      const deliveryHint =
   395→        order.deliveryType === "shahrekord"
   396→          ? "\n\n💡 این سفارش از نوع «تحویل در شهرکرد» است. اگر بسته را حضوری تحویل می‌دهید، می‌توانید بدون کد رهگیری ادامه دهید."
   397→          : "\n\n📦 این سفارش پستی است — لطفاً کد رهگیری را از رسید پست وارد کنید تا مشتری بتواند بسته را در سامانه پست پیگیری کند.";
   398→      await show(
   399→        ctx,
   400→        `📮 <b>تحویل به پست — وارد کردن کد رهگیری</b>\n\n` +
   401→          `🔖 سفارش: <code>${orderNumber}</code>\n` +
   402→          `👤 ${escapeHtml(order.customerName)}\n\n` +
   403→          `لطفاً <b>کد رهگیری پستی</b> را ارسال کنید.\n` +
   404→          `کد رهگیری روی رسید پست نوشته شده است (معمولاً ۱۳ تا ۲۰ رقم).\n\n` +
   405→          `📝 ارقام فارسی هم قابل قبول است.` +
   406→          deliveryHint,
   407→        trackingEntryKb(orderNumber)
   408→      );
   409→      return;
   410→    }
   411→
   412→    // Determine the correct paymentStatus for the new order status.
   413→    // Rule:
   414→    //   - awaiting_payment  → payment must be "pending"  (customer hasn't paid)
   415→    //   - paid              → payment must be "confirmed" (customer clicked "I paid")
   416→    //   - confirmed/preparing/shipped/delivered → payment must be "confirmed"
   417→    //   - cancelled         → leave unchanged (preserve for accounting/audit)
   418→    //
   419→    // This guarantees the site's tracking page never shows an impossible
   420→    // combination like "shipped + pending payment" or "awaiting + confirmed".
   421→    let newPaymentStatus: string | undefined;
   422→    if (status === "awaiting_payment") {
   423→      newPaymentStatus = "pending";
   424→    } else if (status === "cancelled") {
   425→      newPaymentStatus = undefined; // preserve existing
   426→    } else {
   427→      // paid, confirmed, preparing, shipped, delivered
   428→      newPaymentStatus = "confirmed";
   429→    }
   430→
   431→    await withRetry(
   432→      () => db.order.update({
   433→        where: { id: order.id },
   434→        data: {
   435→          orderStatus: status,
   436→          ...(newPaymentStatus !== undefined
   437→            ? { paymentStatus: newPaymentStatus }
   438→            : {}),
   439→        },
   440→      }),
   441→      { label: `updateOrderStatus(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
   442→    );
   443→
   444→    console.log(
   445→      `📊 Status change: ${orderNumber} ${order.orderStatus} → ${status}` +

   498→        select: { id: true, customerName: true, orderStatus: true, trackingCode: true },
   499→      }),
   500→      { label: `findOrderSkip(${orderNumber})` }

   509→      const text = await orderDetailsMessage(orderNumber);

   514→      );
   515→      return;
   516→    }

   828→// ── Text message handler (search + price/desc edit) ──────────────────
   829→export async function handleTextMessage(ctx: Context) {
   830→  const text = ctx.message?.text || "";
   831→  const userId = ctx.from!.id;
   832→
   833→  // Check if in a multi-step flow
   834→  const state = userState.get(userId);
   835→  if (state && state.action === "edit_price") {
   836→    clearState(userId);
   837→    // Convert Persian/Arabic digits to ASCII, then validate.
   838→    // IMPORTANT: We must reject negative numbers, decimals, and any non-digit
   839→    // characters. The old code used .replace(/[^\d]/g, "") which STRIPPED
   840→    // the minus sign, turning "-100" into "100" (a valid positive price!).
   841→    // Now we require the input to be purely digits (after whitespace removal).
   842→    const asciiText = toAsciiDigits(text).replace(/\s/g, "");
   843→    if (!/^\d+$/.test(asciiText)) {
   844→      await ctx.reply(
   845→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید (فقط ارقام، بدون علامت یا کاما).",
   846→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   847→      );
   848→      return;
   849→    }
   850→    const price = parseInt(asciiText, 10);
   851→    if (isNaN(price) || price <= 0) {
   852→      await ctx.reply(
   853→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید.",
   854→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   855→      );
   856→      return;
   857→    }
   858→    if (price > 1_000_000_000) {
   859→      await ctx.reply(
   860→        "❌ قیمت بیش از حد بزرگ است. لطفاً مقدار را بررسی کنید.",
   861→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   862→      );
   863→      return;
   864→    }
   865→    // Fetch old price for the diff message
   866→    const before = await withRetry(
   867→      () => db.product.findUnique({
   868→        where: { slug: state.slug },
   869→        select: { name: true, pricePerKg: true, featured: true },
   870→      }),
   871→      { label: `findProductPrice(${state.slug})` }
   872→    );
   873→    if (!before) {
   874→      await ctx.reply("❌ محصول یافت نشد.", backKb());
   875→      return;
   876→    }
   877→    await withRetry(
   878→      () => db.product.update({
   879→        where: { slug: state.slug },
   880→        data: { pricePerKg: price },
   881→      }),
   882→      { label: `updateProductPrice(${state.slug})`, maxRetries: 5, baseDelayMs: 300 }
   883→    );
   884→    const detailText = await productDetailsMessage(state.slug);
   885→    const diff = price - (before.pricePerKg || 0);
   886→    const diffStr =
   887→      diff === 0
   888→        ? "بدون تغییر"
   889→        : diff > 0
   890→        ? `🔺 افزایش: +${formatToman(diff)}`
   891→        : `🔻 کاهش: ${formatToman(Math.abs(diff))}`;
   892→    await ctx.reply(
   893→      `✅ <b>قیمت محصول به‌روزرسانی شد</b>\n\n` +
   894→        `🍯 ${escapeHtml(before.name)}\n` +
   895→        `💰 قیمت قبلی: ${formatToman(before.pricePerKg)}\n` +
   896→        `💰 قیمت جدید: <b>${formatToman(price)}</b>\n` +
   897→        `📊 ${diffStr}\n\n` +
   898→        `🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   899→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, before.featured) }
   900→    );
   901→    console.log(
   902→      `💰 Product price: ${state.slug} ${before.pricePerKg} → ${price}`
   903→    );
   904→    return;
   905→  }
   906→
   907→  if (state && state.action === "edit_desc") {
   908→    clearState(userId);
   909→    if (!text.trim()) {
   910→      await ctx.reply(
   911→        "❌ توضیحات نمی‌تواند خالی باشد.",
   912→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   913→      );
   914→      return;
   915→    }
   916→    await withRetry(
   917→      () => db.product.update({
   918→        where: { slug: state.slug },
   919→        data: { description: text.trim() },
   920→      }),
   921→      { label: `updateProductDesc(${state.slug})`, maxRetries: 5, baseDelayMs: 300 }
   922→    );
   923→    const product = await withRetry(
   924→      () => db.product.findUnique({
   925→        where: { slug: state.slug },
   926→        select: { name: true, featured: true },
   927→      }),
   928→      { label: `findProductAfterDesc(${state.slug})` }
   929→    );
   930→    const detailText = await productDetailsMessage(state.slug);
   931→    await ctx.reply(
   932→      `✅ توضیحات محصول «${escapeHtml(product?.name || "")}» به‌روزرسانی شد.\n\n🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   933→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, product?.featured || false) }
   934→    );
   935→    console.log(`📝 Product desc updated: ${state.slug}`);
   936→    return;
   937→  }
   938→
   939→  // Tracking code entry flow — when admin is setting an order to "shipped"
   940→  // (تحویل به پست), they must enter the post tracking code.
   941→  // Also handles the "edit tracking code" flow (isEdit=true) where the order
   942→  // is already shipped/delivered and only the trackingCode should be updated.
   943→  if (state && state.action === "enter_tracking") {
   944→    const orderNumber = state.orderNumber;
   945→    const isEdit = state.isEdit === true;
   946→    clearState(userId);
   947→    // Normalize: convert Persian/Arabic digits to ASCII, strip whitespace.
   948→    // Tracking codes are typically all digits (13-20 digits for Iran Post),
   949→    // but we also allow letters (some registered post uses letters).
   950→    const normalized = toAsciiDigits(text).replace(/\s+/g, "");
   951→    // Validation: length 8-30, only alphanumeric
   952→    if (!/^[A-Za-z0-9]{8,30}$/.test(normalized)) {
   953→      await ctx.reply(
   954→        "❌ کد رهگیری نامعتبر است.\n\n" +
   955→          "کد رهگیری باید:\n" +
   956→          "• بین ۸ تا ۳۰ نویسه باشد\n" +
   957→          "• فقط شامل ارقام و حروف انگلیسی باشد\n" +
   958→          "• بدون فاصله یا کاراکتر خاص باشد\n\n" +
   959→          "لطفاً کد رهگیری صحیح را ارسال کنید یا روی «بدون کد رهگیری» بزنید.",
   960→        { parse_mode: "HTML", reply_markup: trackingEntryKb(orderNumber) }
   961→      );
   962→      // Re-set state so the next message is treated as another attempt
   963→      // (preserve isEdit flag)
   964→      userState.set(userId, { action: "enter_tracking", orderNumber, isEdit });
   965→      return;
   966→    }
   967→    try {
   968→      const order = await withRetry(
   969→        () => db.order.findUnique({
   970→          where: { orderNumber },
   971→          select: { id: true, customerName: true, orderStatus: true, trackingCode: true },
   972→        }),
   973→        { label: `findOrderTracking(${orderNumber})` }
   974→      );
   975→      if (!order) {
   976→        await ctx.reply("❌ سفارش یافت نشد.", { reply_markup: backKb(), parse_mode: "HTML" });
   977→        return;
   978→      }
   979→
   980→      if (isEdit) {
   981→        // EDIT mode: only update the tracking code, don't change orderStatus.
   982→        // Guard: only allow if order is shipped or delivered.
   983→        if (order.orderStatus !== "shipped" && order.orderStatus !== "delivered") {
   984→          await ctx.reply(
   985→            `⚠️ این سفارش در وضعیت «${statusLabel(order.orderStatus)}» است و نمی‌توان کد رهگیری آن را ویرایش کرد.`,
   986→            { parse_mode: "HTML", reply_markup: orderActionsKb(orderNumber, order.orderStatus, !!order.trackingCode) }
   987→          );
   988→          return;
   989→        }
   990→        await withRetry(
   991→          () => db.order.update({
   992→            where: { id: order.id },
   993→            data: { trackingCode: normalized },
   994→          }),
   995→          { label: `updateTrackingEdit(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
   996→        );
   997→        console.log(
   998→          `📮 Tracking code updated: ${orderNumber} (was: ${order.trackingCode || "none"} → ${normalized})`
   999→        );
  1000→        const detailText = await orderDetailsMessage(orderNumber);
  1001→        await ctx.reply(
  1002→          `✅ <b>کد رهگیری به‌روزرسانی شد</b>\n\n` +
  1003→            `📋 <code>${orderNumber}</code>\n` +
  1004→            `👤 ${escapeHtml(order.customerName)}\n` +
  1005→            `📊 وضعیت: ${statusLabel(order.orderStatus)}\n` +
  1006→            (order.trackingCode
  1007→              ? `📮 کد قبلی: <code>${escapeHtml(order.trackingCode)}</code>\n`
  1008→              : "") +
  1009→            `📮 کد جدید: <code>${escapeHtml(normalized)}</code>\n\n` +
  1010→            `🌐 مشتری می‌تواند با این کد رهگیری، وضعیت لحظه‌ای بسته را در سامانه پست پیگیری کند.\n\n` +
  1011→            (detailText || ""),
  1012→          { parse_mode: "HTML", reply_markup: orderActionsKb(orderNumber, order.orderStatus, true) }
  1013→        );
  1014→      } else {
  1015→        // NEW SHIPMENT mode: set orderStatus to shipped + trackingCode.
  1016→        await withRetry(
  1017→          () => db.order.update({
  1018→            where: { id: order.id },
  1019→            data: {
  1020→              orderStatus: "shipped",
  1021→              paymentStatus: "confirmed",
  1022→              trackingCode: normalized,
  1023→            },
  1024→          }),
  1025→          { label: `updateOrderTracking(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
  1026→        );
  1027→        console.log(
  1028→          `📊 Status change: ${orderNumber} ${order.orderStatus} → shipped (tracking: ${normalized})`
  1029→        );
  1030→        const detailText = await orderDetailsMessage(orderNumber);
  1031→        await ctx.reply(
  1032→          `✅ <b>سفارش به پست تحویل داده شد</b>\n\n` +
  1033→            `📋 <code>${orderNumber}</code>\n` +
  1034→            `👤 ${escapeHtml(order.customerName)}\n` +
  1035→            `📊 وضعیت قبلی: ${statusLabel(order.orderStatus)}\n` +
  1036→            `📊 وضعیت جدید: ${statusLabel("shipped")}\n` +
  1037→            `📮 کد رهگیری: <code>${escapeHtml(normalized)}</code>\n\n` +
  1038→            `🌐 مشتری می‌تواند با این کد رهگیری، وضعیت لحظه‌ای بسته را در سامانه پست پیگیری کند.\n\n` +
  1039→            (detailText || ""),
  1040→          { parse_mode: "HTML", reply_markup: orderActionsKb(orderNumber, "shipped", true) }
  1041→        );
  1042→      }
  1043→    } catch (e: any) {
  1044→      const errCode = e?.code || e?.errno || "N/A";
  1045→      const errMsg = String(e?.message || e).slice(0, 300);
  1046→      console.error(
  1047→        `❌ enter_tracking FAILED for ${orderNumber} (isEdit=${isEdit}) | code=${errCode} | msg=${errMsg}`
  1048→      );
  1049→      await ctx.reply(
  1050→        `❌ خطا در ثبت کد رهگیری.\n\n⚠️ خطا: <code>${errCode}</code>\n\nلطفاً دوباره تلاش کنید.`,
  1051→        { parse_mode: "HTML", reply_markup: backKb() }
  1052→      );
  1053→    }
  1054→    return;
  1055→  }
  1056→
  1057→  // Default: treat as search
  1058→  const query = text.trim();
  1059→  if (!query) return;
  1060→  try {
  1061→    await ctx.reply("⏳ در حال جستجو...", { parse_mode: "HTML" });
  1062→  } catch {}
  1063→  const msg = await searchMessage(query);
  1064→  const orders = await searchOrders(query);
  1065→  let kb = backKb();
  1066→  if (orders.length > 0) {
  1067→    kb = new InlineKeyboard();
  1068→    for (const o of orders) {
  1069→      kb.text(`${o.orderNumber} | ${o.customerName}`, `o:${o.orderNumber}`).row();
  1070→    }
  1071→    kb.text("🔙 منوی اصلی", "back");
  1072→  }
  1073→  await ctx.reply(msg, {
  1074→    parse_mode: "HTML",
  1075→    reply_markup: kb,
  1076→    disable_web_page_preview: true,
  1077→  });
  1078→}

    17→    17→// Convert an English number to Persian digits
    18→    18→export function toPersianDigits(n: number | string): string {
    19→    19→  const map = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];
    20→    20→  return String(n).replace(/[0-9]/g, (d) => map[parseInt(d, 10)]);
    21→    21→}

    23→    23→// Get start of week (Saturday in Iran calendar)
    24→    24→export function getStartOfWeek(date: Date = new Date()): Date {
    25→    25→  const d = new Date(date);
    26→    26→  const day = d.getDay(); // 0=Sun, 6=Sat
    27→    27→  // In Iran, week starts on Saturday (day 6)
    28→    28→  // To get the most recent Saturday:
    29→    29→  //   if today is Saturday (6), diff = 0
    30→    30→  //   if today is Sunday (0), diff = -1 (1 day back to Saturday)
    31→    31→  //   if today is Monday (1), diff = -2
    32→    32→  //   ...
    33→    33→  //   if today is Friday (5), diff = -6
    34→    34→  let diff = day === 6 ? 0 : -(day + 1);
    35→    35→  d.setDate(d.getDate() + diff);
    36→    36→  d.setHours(0, 0, 0, 0);
    37→    37→  return d;
    38→    38→}
    39→    39→
    40→    40→export function getStartOfMonth(date: Date = new Date()): Date {
    41→    41→  const d = new Date(date);
    42→    42→  d.setDate(1);
    43→    43→  d.setHours(0, 0, 0, 0);
    44→    44→  return d;
    45→    45→}

    49→    49→  d.setDate(d.getDate() - n);
    50→    50→  d.setHours(0, 0, 0, 0);
    51→    51→  return d;
    52→    52→}

    84→    84→export async function computeAgentStats(agentId: string): Promise<AgentStats> {
    85→    85→  const agent = await db.agent.findUnique({
    86→    86→    where: { id: agentId },
    87→    87→    select: {
    88→    88→      id: true,
    89→    89→      commissionRate: true,
    90→    90→      balance: true,
    91→    91→      totalSales: true,
    92→    92→      totalOrders: true,
    93→    93→    },
    94→    94→  });
    95→    95→  if (!agent) throw new Error("Agent not found");
    96→    96→
    97→    97→  // All orders for this agent
    98→    98→  const orders = await db.order.findMany({
    99→    99→    where: { agentId },
   100→   100→    select: {
   101→   101→      id: true,
   102→   102→      orderNumber: true,
   103→   103→      finalAmount: true,
   104→   104→      totalAmount: true,
   105→   105→      uniqueAmount: true,
   106→   106→      orderStatus: true,
   107→   107→      paymentStatus: true,
   108→   108→      createdAt: true,
   109→   109→      items: {
   110→   110→        select: {
   111→   111→          productName: true,
   112→   112→          quantity: true,
   113→   113→          total: true,
   114→   114→        },
   115→   115→      },
   116→   116→    },
   117→   117→    orderBy: { createdAt: "desc" },
   118→   118→  });
   119→   119→
   120→   120→  const totalOrders = orders.length;
   121→   121→  const totalSales = orders.reduce(
   122→   122→    (sum, o) => sum + (o.paymentStatus === "confirmed" ? o.finalAmount : 0),
   123→   123→    0
   124→   124→  );
   125→   125→  const totalCommission = Math.round((totalSales * agent.commissionRate) / 100);
   126→   126→  const pendingOrders = orders.filter(
   127→   127→    (o) =>
   128→   128→      o.orderStatus === "awaiting_payment" || o.orderStatus === "paid"
   129→   129→  ).length;
   130→   130→
   131→   131→  // This week's sales (confirmed payments only)
   132→   132→  const weekStart = getStartOfWeek();
   133→   133→  const monthStart = getStartOfMonth();
   134→   134→  const lastWeekStart = new Date(weekStart);
   135→   135→  lastWeekStart.setDate(lastWeekStart.getDate() - 7);
   136→   136→  const lastMonthStart = new Date(monthStart);
   137→   137→  lastMonthStart.setMonth(lastMonthStart.getMonth() - 1);
   138→   138→
   139→   139→  const thisWeekSales = orders
   140→   140→    .filter(
   141→   141→      (o) =>
   142→   142→        o.paymentStatus === "confirmed" && new Date(o.createdAt) >= weekStart
   143→   143→    )
   144→   144→    .reduce((s, o) => s + o.finalAmount, 0);
   145→   145→  const thisMonthSales = orders
   146→   146→    .filter(
   147→   147→      (o) =>
   148→   148→        o.paymentStatus === "confirmed" && new Date(o.createdAt) >= monthStart
   149→   149→    )
   150→   150→    .reduce((s, o) => s + o.finalAmount, 0);
   151→   151→  const lastWeekSales = orders
   152→   152→    .filter(
   153→   153→      (o) =>
   154→   154→        o.paymentStatus === "confirmed" &&
   155→   155→        new Date(o.createdAt) >= lastWeekStart &&
   156→   156→        new Date(o.createdAt) < weekStart
   157→   157→    )
   158→   158→    .reduce((s, o) => s + o.finalAmount, 0);
   159→   159→  const lastMonthSales = orders
   160→   160→    .filter(
   161→   161→      (o) =>
   162→   162→        o.paymentStatus === "confirmed" &&
   163→   163→        new Date(o.createdAt) >= lastMonthStart &&
   164→   164→        new Date(o.createdAt) < monthStart
   165→   165→    )
   166→   166→    .reduce((s, o) => s + o.finalAmount, 0);
   167→   167→
   168→   168→  const weekGrowthPct =
   169→   169→    lastWeekSales > 0
   170→   170→      ? Math.round(((thisWeekSales - lastWeekSales) / lastWeekSales) * 100)
   171→   171→      : thisWeekSales > 0
   172→   172→      ? 100
   173→   173→      : null;
   174→   174→  const monthGrowthPct =
   175→   175→    lastMonthSales > 0
   176→   176→      ? Math.round(((thisMonthSales - lastMonthSales) / lastMonthSales) * 100)
   177→   177→      : thisMonthSales > 0
   178→   178→      ? 100
   179→   179→      : null;
   180→   180→
   181→   181→  // Weekly series (last 8 weeks)
   182→   182→  const weeklySeries: { label: string; sales: number; orders: number }[] = [];
   183→   183→  for (let i = 7; i >= 0; i--) {
   184→   184→    const wStart = new Date(weekStart);
   185→   185→    wStart.setDate(wStart.getDate() - i * 7);
   186→   186→    const wEnd = new Date(wStart);
   187→   187→    wEnd.setDate(wEnd.getDate() + 7);
   188→   188→    const weekOrders = orders.filter((o) => {
   189→   189→      const d = new Date(o.createdAt);
   190→   190→      return d >= wStart && d < wEnd;
   191→   191→    });
   192→   192→    const sales = weekOrders
   193→   193→      .filter((o) => o.paymentStatus === "confirmed")
   194→   194→      .reduce((s, o) => s + o.finalAmount, 0);
   195→   195→    weeklySeries.push({
   196→   196→      label: `هفته ${toPersianDigits(8 - i)}`,
   197→   197→      sales,
   198→   198→      orders: weekOrders.length,
   199→   199→    });
   200→   200→  }
   201→   201→
   202→   202→  // Monthly series (last 6 months)
   203→   203→  const monthlySeries: { label: string; sales: number; orders: number }[] = [];
   204→   204→  const monthNames = [
   205→   205→    "فروردین", "اردیبهشت", "خرداد", "تیر", "مرداد", "شهریور",
   206→   206→    "مهر", "آبان", "آذر", "دی", "بهمن", "اسفند"
   207→   207→  ];
   208→   208→  // Approximate Jalali month using simple offset (good enough for labels)
   209→   209→  for (let i = 5; i >= 0; i--) {
   210→   210→    const mStart = new Date(monthStart);
   211→   211→    mStart.setMonth(mStart.getMonth() - i);
   212→   212→    const mEnd = new Date(mStart);
   213→   213→    mEnd.setMonth(mEnd.getMonth() + 1);
   214→   214→    const monthOrders = orders.filter((o) => {
   215→   215→      const d = new Date(o.createdAt);
   216→   216→      return d >= mStart && d < mEnd;
   217→   217→    });
   218→   218→    const sales = monthOrders
   219→   219→      .filter((o) => o.paymentStatus === "confirmed")
   220→   220→      .reduce((s, o) => s + o.finalAmount, 0);
   221→   221→    // Approximate Jalali month index (Gregorian month - 3, mod 12, then +1 if past March 21)
   222→   222→    let jIdx = (mStart.getMonth() - 2 + 12) % 12;
   223→   223→    if (mStart.getMonth() < 2) jIdx = (mStart.getMonth() + 10) % 12;
   224→   224→    monthlySeries.push({
   225→   225→      label: monthNames[jIdx],
   226→   226→      sales,
   227→   227→      orders: monthOrders.length,
   228→   228→    });
   229→   229→  }
   230→   230→
   231→   231→  // Order status distribution
   232→   232→  const statusCounts: Record<string, number> = {};
   233→   233→  for (const o of orders) {
   234→   234→    statusCounts[o.orderStatus] = (statusCounts[o.orderStatus] || 0) + 1;
   235→   235→  }
   236→   236→  const statusLabels: Record<string, string> = {
   237→   237→    awaiting_payment: "در انتظار پرداخت",
   238→   238→    paid: "پرداخت شده",
   239→   239→    confirmed: "تأیید شده",
   240→   240→    preparing: "در حال آماده‌سازی",
   241→   241→    shipped: "ارسال شده",
   242→   242→    delivered: "تحویل داده شده",
   243→   243→    cancelled: "لغو شده",
   244→   244→  };
   245→   245→  const orderStatusDistribution = Object.entries(statusCounts).map(
   246→   246→    ([status, count]) => ({
   247→   247→      status: statusLabels[status] || status,
   248→   248→      count,
   249→   249→    })
   250→   250→  );
   251→   251→
   252→   252→  // Top products (by total quantity)
   253→   253→  const productAgg: Record<string, { quantity: number; total: number }> = {};
   254→   254→  for (const o of orders) {
   255→   255→    for (const item of o.items) {
   256→   256→      if (!productAgg[item.productName]) {
   257→   257→        productAgg[item.productName] = { quantity: 0, total: 0 };
   258→   258→      }
   259→   259→      productAgg[item.productName].quantity += item.quantity;
   260→   260→      productAgg[item.productName].total += item.total;
   261→   261→    }
   262→   262→  }
   263→   263→  const topProducts = Object.entries(productAgg)
   264→   264→    .map(([productName, v]) => ({ productName, ...v }))
   265→   265→    .sort((a, b) => b.quantity - a.quantity)
   266→   266→    .slice(0, 5);
   267→   267→
   268→   268→  // Recent orders (last 5)
   269→   269→  const recentOrders = orders.slice(0, 5).map((o) => ({
   270→   270→    id: o.id,
   271→   271→    orderNumber: o.orderNumber,
   272→   272→    finalAmount: o.finalAmount,
   273→   273→    orderStatus: statusLabels[o.orderStatus] || o.orderStatus,
   274→   274→    paymentStatus: o.paymentStatus === "confirmed" ? "تأیید شده" : "در انتظار",
   275→   275→    createdAt: o.createdAt.toISOString(),
   276→   276→  }));
   277→   277→
   278→   278→  return {
   279→   279→    totalOrders,
   280→   280→    totalSales,
   281→   281→    totalCommission,
   282→   282→    balance: agent.balance,
   283→   283→    commissionRate: agent.commissionRate,
   284→   284→    pendingOrders,
   285→   285→    thisWeekSales,
   286→   286→    thisMonthSales,
   287→   287→    weekGrowthPct,
   288→   288→    monthGrowthPct,
   289→   289→    weeklySeries,
   290→   290→    monthlySeries,
   291→   291→    orderStatusDistribution,
   292→   292→    topProducts,
   293→   293→    recentOrders,
   294→   294→  };
   295→   295→}

   372→   372→      id: true,
   373→   373→      name: true,
   374→   374→      storeName: true,

   379→   379→      totalSales: true,
   380→   380→      totalOrders: true,
   381→   381→      createdAt: true,
   382→   382→    },
   383→   383→  });
   384→   384→  const totalAgents = agents.length;
   385→   385→  const activeAgents = agents.filter((a) => a.status === "active").length;
   386→   386→  const pendingAgents = agents.filter((a) => a.status === "pending").length;
   387→   387→  const blockedAgents = agents.filter((a) => a.status === "blocked").length;
   388→   388→
   389→   389→  // All orders (lifetime, for series + recent activity)
   390→   390→  const orders = await db.order.findMany({
   391→   391→    select: {
   392→   392→      id: true,
   393→   393→      orderNumber: true,
   394→   394→      customerName: true,
   395→   395→      finalAmount: true,
   396→   396→      orderType: true,
   397→   397→      orderStatus: true,
   398→   398→      paymentStatus: true,
   399→   399→      createdAt: true,
   400→   400→      agent: { select: { id: true, name: true, storeName: true } },
   401→   401→      items: { select: { productName: true, quantity: true, total: true } },
   402→   402→    },
   403→   403→    orderBy: { createdAt: "desc" },
   404→   404→  });
   405→   405→  const totalOrders = orders.length;
   406→   406→  const agentOrders = orders.filter((o) => o.orderType === "agent").length;
   407→   407→  const customerOrders = orders.filter((o) => o.orderType === "customer").length;
   408→   408→
   409→   409→  // Commission paid (lifetime — based on agent's stored totalSales)
   410→   410→  const totalCommissionPaid = agents.reduce(
   411→   411→    (s, a) => s + Math.round((a.totalSales * a.commissionRate) / 100),
   412→   412→    0
   413→   413→  );
   414→   414→
   415→   415→  // Period filter — affects revenue totals, top products, top agents,
   416→   416→  // status distribution. Snapshot stats (thisWeek/Month) stay lifetime.
   417→   417→  const now = new Date();
   418→   418→  const periodStart = new Date(now);
   419→   419→  periodStart.setDate(periodStart.getDate() - (periodDays - 1));
   420→   420→  periodStart.setHours(0, 0, 0, 0);
   421→   421→
   422→   422→  const periodOrders = orders.filter(
   423→   423→    (o) => new Date(o.createdAt) >= periodStart
   424→   424→  );
   425→   425→  const periodConfirmed = periodOrders.filter(
   426→   426→    (o) => o.paymentStatus === "confirmed"
   427→   427→  );
   428→   428→  const totalRevenue = periodConfirmed.reduce((s, o) => s + o.finalAmount, 0);
   429→   429→  const agentRevenue = periodConfirmed
   430→   430→    .filter((o) => o.orderType === "agent")
   431→   431→    .reduce((s, o) => s + o.finalAmount, 0);
   432→   432→  const customerRevenue = totalRevenue - agentRevenue;
   433→   433→
   434→   434→  // This period snapshots — always "this week" / "this month" regardless
   435→   435→  // of the period filter
   436→   436→  const weekStart = getStartOfWeek();
   437→   437→  const monthStart = getStartOfMonth();
   438→   438→  const lastWeekStart = new Date(weekStart);
   439→   439→  lastWeekStart.setDate(lastWeekStart.getDate() - 7);
   440→   440→  const lastMonthStart = new Date(monthStart);
   441→   441→  lastMonthStart.setMonth(lastMonthStart.getMonth() - 1);
   442→   442→
   443→   443→  const confirmedOrders = orders.filter((o) => o.paymentStatus === "confirmed");
   444→   444→  const thisWeekRevenue = confirmedOrders
   445→   445→    .filter((o) => new Date(o.createdAt) >= weekStart)
   446→   446→    .reduce((s, o) => s + o.finalAmount, 0);
   447→   447→  const thisMonthRevenue = confirmedOrders
   448→   448→    .filter((o) => new Date(o.createdAt) >= monthStart)
   449→   449→    .reduce((s, o) => s + o.finalAmount, 0);
   450→   450→  const lastWeekRevenue = confirmedOrders
   451→   451→    .filter(
   452→   452→      (o) =>
   453→   453→        new Date(o.createdAt) >= lastWeekStart &&
   454→   454→        new Date(o.createdAt) < weekStart
   455→   455→    )
   456→   456→    .reduce((s, o) => s + o.finalAmount, 0);
   457→   457→  const lastMonthRevenue = confirmedOrders
   458→   458→    .filter(
   459→   459→      (o) =>
   460→   460→        new Date(o.createdAt) >= lastMonthStart &&
   461→   461→        new Date(o.createdAt) < monthStart
   462→   462→    )
   463→   463→    .reduce((s, o) => s + o.finalAmount, 0);
   464→   464→

    53→// Each flow stores { action, payload }. Cleared on any navigation.

    62→// ── Helper: extract callback data from ctx ───────────────────────────
    63→// grammy's ctx.match is a RegExpMatchArray when using regex patterns.
    64→// We need the raw callback_data string instead, which is in ctx.callbackQuery.data.
    65→const cbData = (ctx: Context): string => ctx.callbackQuery?.data || "";
    66→
    67→// Extract the "payload" portion of a callback data string (everything after the first colon).
    68→// Example: "o:HN-12345" → "HN-12345"; "oss:HN-12345:confirmed" → "HN-12345:confirmed"
    69→const cbPayload = (ctx: Context): string => {
    70→  const data = cbData(ctx);
    71→  const idx = data.indexOf(":");
    72→  return idx >= 0 ? data.slice(idx + 1) : "";
    73→};
    74→
    75→// Extract page number from a paginated callback like "today:0", "all:2", "cust:1"
    76→const cbPage = (ctx: Context): number => {
    77→  const data = cbData(ctx);
    78→  const parts = data.split(":");
    79→  return Number(parts[1] || 0);
    80→};

    82→// ── Helper: show text (edit message if possible, else reply) ─────────
    83→async function show(ctx: Context, text: string, keyboard?: any) {
    84→  const opts: any = {
    85→    parse_mode: "HTML",
    86→    disable_web_page_preview: true,
    87→  };
    88→  if (keyboard) opts.reply_markup = keyboard;
    89→  try {
    90→    await ctx.editMessageText(text, opts);
    91→  } catch {
    92→    try {
    93→      await ctx.reply(text, opts);
    94→    } catch (e) {
    95→      console.error("show() reply failed:", e);
    96→    }
    97→  }
    98→}

   100→// ── Access control middleware ────────────────────────────────────────
   101→export async function accessControl(ctx: Context, next: () => Promise<void>) {
   102→  if (!ctx.from) return;
   103→  if (ctx.from.id !== ADMIN_ID) {
   104→    try {
   105→      await ctx.reply(
   106→        "⛔ شما دسترسی به این ربات ندارید.\n\n" +
   107→          "این ربات اختصاصاً برای مدیریت فروشگاه «سرزمین عسل» طراحی شده است."
   108→      );
   109→    } catch {}
   110→    return;
   111→  }
   112→  return next();
   113→}
   114→
   115→// Auto-answer callback queries to remove loading spinner.
   116→// ALSO: clear any pending multi-step flow state (edit_price / edit_desc) on
   117→// every button click. This prevents a stale state from causing the admin's next
   118→// text message to be misinterpreted as a price/desc edit for the OLD product.
   119→// (Handlers that START a flow — handleProductEditPrice, handleProductEditDesc —
   120→// re-set the state AFTER this middleware clears it, so the flow still works.)
   121→export async function answerCallbacks(ctx: Context, next: () => Promise<void>) {
   122→  if (ctx.callbackQuery) {
   123→    try {
   124→      await ctx.answerCallbackQuery();
   125→    } catch {}
   126→    // Clear any pending edit flow — the admin clicked a button, so they're
   127→    // navigating away from the text-input flow. If they clicked "edit price"
   128→    // or "edit desc", the handler will re-set the state after this clears it.
   129→    if (ctx.from) clearState(ctx.from.id);
   130→  }
   131→  return next();
   132→}

   138→  const now = new Date();

   140→  todayStart.setHours(0, 0, 0, 0);

   142→    db.order.count({ where: { createdAt: { gte: todayStart } } }),
   143→    db.order.count({ where: { orderStatus: "paid" } }),
   144→  ]);
   145→  return { todayCount, verifyCount };
   146→}

   318→// ── Set order status ─────────────────────────────────────────────────
   319→// This is the admin's primary manual action — changing an order's status.
   320→// It handles the full status workflow: awaiting_payment → paid → confirmed
   321→// → preparing → shipped → delivered (plus cancelled as a side state).
   322→// When the admin advances to "confirmed" or later, payment is also marked
   323→// as confirmed. When moved back to "awaiting_payment", payment resets to pending.
   324→export async function handleSetOrderStatus(ctx: Context) {
   325→  // pattern: oss:<orderNumber>:<status>  (orderNumber has no colon, so parts[1]=orderNumber, parts[2]=status)
   326→  const parts = cbData(ctx).split(":");
   327→  const orderNumber = parts[1];
   328→  const status = parts[2];
   329→
   330→  // Validate the status is a known value
   331→  const validStatuses = [
   332→    "awaiting_payment", "paid", "confirmed", "preparing", "shipped", "delivered", "cancelled",
   333→  ];
   334→  if (!validStatuses.includes(status)) {
   335→    await show(ctx, "❌ وضعیت نامعتبر است.", backKb());
   336→    return;
   337→  }
   338→
   339→  try {
   340→    // Show a "processing" indicator so the admin knows the click was received
   341→    try {
   342→      await ctx.editMessageText("⏳ در حال به‌روزرسانی وضعیت...", { parse_mode: "HTML" });
   343→    } catch {}
   344→
   345→    const order = await withRetry(
   346→      () => db.order.findUnique({
   347→        where: { orderNumber },
   348→        select: { id: true, customerName: true, orderStatus: true },
   349→      }),
   350→      { label: `findOrder(${orderNumber})` }
   351→    );
   352→    if (!order) {
   353→      await show(ctx, "❌ سفارش یافت نشد.", backKb());
   354→      return;
   355→    }
   356→
   357→    // If status hasn't changed, no-op (avoid unnecessary writes)
   358→    if (order.orderStatus === status) {
   359→      const text = await orderDetailsMessage(orderNumber);
   360→      await show(ctx, `ℹ️ وضعیت سفارش از قبل «${statusLabel(status)}» بود.\n\n${text || ""}`, orderActionsKb(orderNumber, status));
   361→      return;
   362→    }
   363→
   364→    // Determine the correct paymentStatus for the new order status.
   365→    // Rule:
   366→    //   - awaiting_payment  → payment must be "pending"  (customer hasn't paid)
   367→    //   - paid              → payment must be "confirmed" (customer clicked "I paid")
   368→    //   - confirmed/preparing/shipped/delivered → payment must be "confirmed"
   369→    //   - cancelled         → leave unchanged (preserve for accounting/audit)
   370→    //
   371→    // This guarantees the site's tracking page never shows an impossible
   372→    // combination like "shipped + pending payment" or "awaiting + confirmed".
   373→    let newPaymentStatus: string | undefined;
   374→    if (status === "awaiting_payment") {
   375→      newPaymentStatus = "pending";
   376→    } else if (status === "cancelled") {
   377→      newPaymentStatus = undefined; // preserve existing
   378→    } else {
   379→      // paid, confirmed, preparing, shipped, delivered
   380→      newPaymentStatus = "confirmed";
   381→    }
   382→
   383→    await withRetry(
   384→      () => db.order.update({
   385→        where: { id: order.id },
   386→        data: {
   387→          orderStatus: status,
   388→          ...(newPaymentStatus !== undefined
   389→            ? { paymentStatus: newPaymentStatus }
   390→            : {}),
   391→        },
   392→      }),
   393→      { label: `updateOrderStatus(${orderNumber})`, maxRetries: 5, baseDelayMs: 300 }
   394→    );
   395→
   396→    console.log(
   397→      `📊 Status change: ${orderNumber} ${order.orderStatus} → ${status}` +
   398→        (newPaymentStatus !== undefined
   399→          ? ` (payment: ${newPaymentStatus})`
   400→          : "")
   401→    );
   402→
   403→    const text = await orderDetailsMessage(orderNumber);
   404→    const msg =
   405→      `✅ <b>وضعیت سفارش به‌روزرسانی شد</b>\n\n` +
   406→      `📋 <code>${orderNumber}</code>\n` +
   407→      `👤 ${escapeHtml(order.customerName)}\n` +
   408→      `📊 وضعیت قبلی: ${statusLabel(order.orderStatus)}\n` +
   409→      `📊 وضعیت جدید: ${statusLabel(status)}\n\n` +
   410→      (text || "");
   411→    await show(ctx, msg, orderActionsKb(orderNumber, status));
   412→  } catch (e: any) {
   413→    // Log the FULL error details so we can diagnose the root cause
   414→    const errCode = e?.code || e?.errno || "N/A";
   415→    const errMsg = String(e?.message || e).slice(0, 300);
   416→    const errStack = e?.stack ? String(e.stack).split("\n").slice(0, 3).join(" | ") : "";
   417→    console.error(
   418→      `❌ handleSetOrderStatus FAILED for ${orderNumber} → ${status} | ` +
   419→      `code=${errCode} | msg=${errMsg}` +
   420→      (errStack ? ` | stack=${errStack}` : "")
   421→    );
   422→    await show(
   423→      ctx,
   424→      `❌ <b>خطا در به‌روزرسانی وضعیت سفارش</b>\n\n` +
   425→        `📋 سفارش: <code>${orderNumber}</code>\n` +
   426→        `📊 وضعیت هدف: ${statusLabel(status)}\n` +
   427→        `⚠️ خطا: <code>${errCode}</code>\n\n` +
   428→        `لطفاً دوباره تلاش کنید. اگر خطا تکرار شد، چند ثانیه صبر کنید و دوباره امتحان کنید.`,
   429→      backKb()
   430→    );
   431→  }
   432→}

   612→// ── Text message handler (search + price/desc edit) ──────────────────
   613→export async function handleTextMessage(ctx: Context) {
   614→  const text = ctx.message?.text || "";
   615→  const userId = ctx.from!.id;
   616→
   617→  // Check if in a multi-step flow
   618→  const state = userState.get(userId);
   619→  if (state && state.action === "edit_price") {
   620→    clearState(userId);
   621→    // Convert Persian/Arabic digits to ASCII, then validate.
   622→    // IMPORTANT: We must reject negative numbers, decimals, and any non-digit
   623→    // characters. The old code used .replace(/[^\d]/g, "") which STRIPPED
   624→    // the minus sign, turning "-100" into "100" (a valid positive price!).
   625→    // Now we require the input to be purely digits (after whitespace removal).
   626→    const asciiText = toAsciiDigits(text).replace(/\s/g, "");
   627→    if (!/^\d+$/.test(asciiText)) {
   628→      await ctx.reply(
   629→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید (فقط ارقام، بدون علامت یا کاما).",
   630→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   631→      );
   632→      return;
   633→    }
   634→    const price = parseInt(asciiText, 10);
   635→    if (isNaN(price) || price <= 0) {
   636→      await ctx.reply(
   637→        "❌ قیمت نامعتبر است. لطفاً یک عدد صحیح مثبت وارد کنید.",
   638→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   639→      );
   640→      return;
   641→    }
   642→    if (price > 1_000_000_000) {
   643→      await ctx.reply(
   644→        "❌ قیمت بیش از حد بزرگ است. لطفاً مقدار را بررسی کنید.",
   645→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   646→      );
   647→      return;
   648→    }
   649→    // Fetch old price for the diff message
   650→    const before = await db.product.findUnique({
   651→      where: { slug: state.slug },
   652→      select: { name: true, pricePerKg: true, featured: true },
   653→    });
   654→    if (!before) {
   655→      await ctx.reply("❌ محصول یافت نشد.", backKb());
   656→      return;
   657→    }
   658→    await db.product.update({
   659→      where: { slug: state.slug },
   660→      data: { pricePerKg: price },
   661→    });
   662→    const detailText = await productDetailsMessage(state.slug);
   663→    const diff = price - (before.pricePerKg || 0);
   664→    const diffStr =
   665→      diff === 0
   666→        ? "بدون تغییر"
   667→        : diff > 0
   668→        ? `🔺 افزایش: +${formatToman(diff)}`
   669→        : `🔻 کاهش: ${formatToman(Math.abs(diff))}`;
   670→    await ctx.reply(
   671→      `✅ <b>قیمت محصول به‌روزرسانی شد</b>\n\n` +
   672→        `🍯 ${escapeHtml(before.name)}\n` +
   673→        `💰 قیمت قبلی: ${formatToman(before.pricePerKg)}\n` +
   674→        `💰 قیمت جدید: <b>${formatToman(price)}</b>\n` +
   675→        `📊 ${diffStr}\n\n` +
   676→        `🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   677→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, before.featured) }
   678→    );
   679→    console.log(
   680→      `💰 Product price: ${state.slug} ${before.pricePerKg} → ${price}`
   681→    );
   682→    return;
   683→  }
   684→
   685→  if (state && state.action === "edit_desc") {
   686→    clearState(userId);
   687→    if (!text.trim()) {
   688→      await ctx.reply(
   689→        "❌ توضیحات نمی‌تواند خالی باشد.",
   690→        { reply_markup: productActionsKb(state.slug), parse_mode: "HTML" }
   691→      );
   692→      return;
   693→    }
   694→    await db.product.update({
   695→      where: { slug: state.slug },
   696→      data: { description: text.trim() },
   697→    });
   698→    const product = await db.product.findUnique({
   699→      where: { slug: state.slug },
   700→      select: { name: true, featured: true },
   701→    });
   702→    const detailText = await productDetailsMessage(state.slug);
   703→    await ctx.reply(
   704→      `✅ توضیحات محصول «${escapeHtml(product?.name || "")}» به‌روزرسانی شد.\n\n🌐 این تغییر بلافاصله در سایت اعمال می‌شود.\n\n${detailText || ""}`,
   705→      { parse_mode: "HTML", reply_markup: productActionsKb(state.slug, product?.featured || false) }
   706→    );
   707→    console.log(`📝 Product desc updated: ${state.slug}`);
   708→    return;
   709→  }
   710→
   711→  // Default: treat as search
   712→  const query = text.trim();
   713→  if (!query) return;
   714→  try {
   715→    await ctx.reply("⏳ در حال جستجو...", { parse_mode: "HTML" });
   716→  } catch {}
   717→  const msg = await searchMessage(query);
   718→  const orders = await searchOrders(query);
   719→  let kb = backKb();
   720→  if (orders.length > 0) {
   721→    kb = new InlineKeyboard();
   722→    for (const o of orders) {
   723→      kb.text(`${o.orderNumber} | ${o.customerName}`, `o:${o.orderNumber}`).row();
   724→    }
   725→    kb.text("🔙 منوی اصلی", "back");
   726→  }
   727→  await ctx.reply(msg, {
   728→    parse_mode: "HTML",
   729→    reply_markup: kb,
   730→    disable_web_page_preview: true,
   731→  });
   732→}
